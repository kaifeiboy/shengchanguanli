```python
import time
from typing import Dict, List, Optional
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class MultiModelOcrEngine:
    def __init__(self):
        self.models = {}
        self.model_load_status = {}
        self._initialize_models()
    
    def _initialize_models(self):
        """初始化多个OCR模型"""
        # RapidOCR
        try:
            from rapidocr_onnxruntime import RapidOCR
            self.models['rapidocr'] = RapidOCR()
            self.model_load_status['rapidocr'] = True
            logger.info("RapidOCR loaded successfully")
        except Exception as e:
            logger.error(f"Failed to load RapidOCR: {e}")
            self.model_load_status['rapidocr'] = False
        
        # PaddleOCR
        try:
            from paddleocr import PaddleOCR
            self.models['paddleocr'] = PaddleOCR(
                use_angle_cls=True, 
                lang='ch',
                show_log=False,
                use_gpu=False
            )
            self.model_load_status['paddleocr'] = True
            logger.info("PaddleOCR loaded successfully")
        except Exception as e:
            logger.error(f"Failed to load PaddleOCR: {e}")
            self.model_load_status['paddleocr'] = False
        
        # Tesseract (可选)
        try:
            import pytesseract
            self.models['tesseract'] = pytesseract
            self.model_load_status['tesseract'] = True
            logger.info("Tesseract loaded successfully")
        except Exception as e:
            logger.error(f"Failed to load Tesseract: {e}")
            self.model_load_status['tesseract'] = False
    
    def recognize_single_model(self, model_name: str, image_path: str) -> Optional[dict]:
        """使用单个模型识别"""
        if model_name not in self.models or not self.model_load_status[model_name]:
            return None
        
        try:
            model = self.models[model_name]
            start_time = time.time()
            
            if model_name == 'rapidocr':
                result = model(image_path)
                if result and len(result) > 0:
                    ocr_result = result[0]
                    text = "\n".join([line[1][0] for line in ocr_result])
                    confidence = sum([line[1][1] for line in ocr_result]) / len(ocr_result) if ocr_result else 0
                    return {
                        'text': text,
                        'confidence': confidence,
                        'details': ocr_result,
                        'model': model_name,
                        'time': time.time() - start_time
                    }
            
            elif model_name == 'paddleocr':
                result = model.ocr(image_path, cls=True)
                if result and len(result) > 0 and result[0]:
                    ocr_result = result[0]
                    text = "\n".join([line[1][0] for line in ocr_result])
                    confidence = sum([line[1][1] for line in ocr_result]) / len(ocr_result) if ocr_result else 0
                    return {
                        'text': text,
                        'confidence': confidence,
                        'details': ocr_result,
                        'model': model_name,
                        'time': time.time() - start_time
                    }
            
            elif model_name == 'tesseract':
                from PIL import Image
                image = Image.open(image_path)
                text = model.image_to_string(image, lang='eng+chi_sim')
                return {
                    'text': text.strip(),
                    'confidence': 0.8,  # Tesseract不提供置信度，使用默认值
                    'details': None,
                    'model': model_name,
                    'time': time.time() - start_time
                }
        
        except Exception as e:
            logger.error(f"Error in {model_name}: {e}")
            return None
        
        return None
    
    def recognize_all(self, image_path: str) -> Dict[str, dict]:
        """使用所有可用模型识别"""
        results = {}
        
        for model_name in self.models.keys():
            if self.model_load_status[model_name]:
                result = self.recognize_single_model(model_name, image_path)
                if result:
                    results[model_name] = result
        
        return results
    
    def get_available_models(self) -> List[str]:
        """获取可用的模型列表"""
        return [name for name, status in self.model_load_status.items() if status]
```

#### 1.4 实现基础融合引擎
**文件**: `ocr_engine/fusion/basic_fusion.py`

```python
from typing import Dict, List
import re
from collections import Counter

class BasicFusionEngine:
    def fuse_results(self, results: Dict[str, dict]) -> dict:
        """基础融合策略"""
        if not results:
            return {'text': '', 'confidence': 0, 'method': 'none'}
        
        # 如果只有一个结果，直接返回
        if len(results) == 1:
            only_result = list(results.values())[0]
            return {
                'text': only_result['text'],
                'confidence': only_result['confidence'],
                'method': 'single_model',
                'source': only_result['model']
            }
        
        # 多个结果，应用融合策略
        return self.weighted_fusion(results)
    
    def weighted_fusion(self, results: Dict[str, dict]) -> dict:
        """基于置信度的加权融合"""
        # 计算权重
        total_confidence = sum(r['confidence'] for r in results.values())
        if total_confidence == 0:
            return self.majority_voting(results)
        
        # 加权文本合并
        weighted_texts = []
        for model_name, result in results.items():
            weight = result['confidence'] / total_confidence
            weighted_texts.append((result['text'], weight, model_name))
        
        # 选择置信度最高的结果作为基础
        best_result = max(results.values(), key=lambda x: x['confidence'])
        
        return {
            'text': best_result['text'],
            'confidence': best_result['confidence'],
            'method': 'weighted_fusion',
            'source': best_result['model'],
            'all_results': {k: v['text'] for k, v in results.items()}
        }
    
    def majority_voting(self, results: Dict[str, dict]) -> dict:
        """多数投票策略"""
        texts = [r['text'] for r in results.values()]
        
        # 简化文本比较
        simplified_texts = [self.simplify_text(t) for t in texts]
        
        # 找出最相似的文本组
        best_text = self.find_most_common_text(simplified_texts, texts)
        
        # 平均置信度
        avg_confidence = sum(r['confidence'] for r in results.values()) / len(results)
        
        return {
            'text': best_text,
            'confidence': avg_confidence,
            'method': 'majority_voting',
            'all_texts': texts
        }
    
    def simplify_text(self, text: str) -> str:
        """简化文本用于比较"""
        # 移除空格、换行，转小写
        simplified = re.sub(r'[\s\n]', '', text.lower())
        # 移除特殊字符
        simplified = re.sub(r'[^\w\u4e00-\u9fff]', '', simplified)
        return simplified
    
    def find_most_common_text(self, simplified_texts: List[str], original_texts: List[str]) -> str:
        """找出最常见的文本"""
        if not simplified_texts:
            return ""
        
        # 找出最相似的文本对
        max_similarity = 0
        best_pair = (0, 0)
        
        for i in range(len(simplified_texts)):
            for j in range(i + 1, len(simplified_texts)):
                similarity = self.calculate_similarity(simplified_texts[i], simplified_texts[j])
                if similarity > max_similarity:
                    max_similarity = similarity
                    best_pair = (i, j)
        
        # 返回相似度最高的文本中较长的那个
        i, j = best_pair
        if len(original_texts[i]) >= len(original_texts[j]):
            return original_texts[i]
        return original_texts[j]
    
    def calculate_similarity(self, text1: str, text2: str) -> float:
        """计算文本相似度"""
        if not text1 or not text2:
            return 0.0
        
        # 简单的字符重叠度
        set1 = set(text1)
        set2 = set(text2)
        
        if not set1 or not set2:
            return 0.0
        
        intersection = len(set1 & set2)
        union = len(set1 | set2)
        
        return intersection / union if union > 0 else 0.0
```

```python
import cv2
import numpy as np
from typing import Dict

class ImageAnalyzer:
    def analyze(self, image_path: str) -> Dict[str, float]:
        """分析图像特征"""
        image = cv2.imread(image_path)
        if image is None:
            return {}
        
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        
        features = {
            'noise_level': self.estimate_noise(gray),
            'brightness': self.estimate_brightness(gray),
            'contrast': self.estimate_contrast(gray),
            'blur_level': self.estimate_blur(gray),
            'sharpness': self.estimate_sharpness(gray)
        }
        
        return features
    
    def estimate_noise(self, gray_image: np.ndarray) -> float:
        """估计噪声水平"""
        # 使用拉普拉斯算子估计噪声
        laplacian = cv2.Laplacian(gray_image, cv2.CV_64F)
        noise_level = np.std(laplacian) / 255.0
        return min(noise_level, 1.0)  # 归一化到0-1
    
    def estimate_brightness(self, gray_image: np.ndarray) -> float:
        """估计亮度"""
        return np.mean(gray_image) / 255.0
    
    def estimate_contrast(self, gray_image: np.ndarray) -> float:
        """估计对比度"""
        return np.std(gray_image) / 128.0  # 归一化
    
    def estimate_blur(self, gray_image: np.ndarray) -> float:
        """估计模糊程度 (值越大越模糊)"""
        # 使用拉普拉斯方差
        laplacian_var = cv2.Laplacian(gray_image, cv2.CV_64F).var()
        # 归一化: 值越小越模糊
        blur_level = 1.0 - min(laplacian_var / 1000.0, 1.0)
        return blur_level
    
    def estimate_sharpness(self, gray_image: np.ndarray) -> float:
        """估计清晰度"""
        # 使用Sobel算子
        sobel_x = cv2.Sobel(gray_image, cv2.CV_64F, 1, 0, ksize=3)
        sobel_y = cv2.Sobel(gray_image, cv2.CV_64F, 0, 1, ksize=3)
        sharpness = np.mean(np.sqrt(sobel_x**2 + sobel_y**2)) / 255.0
        return min(sharpness, 1.0)
```

#### 2.2 自适应预处理
**文件**: `ocr_engine/preprocessing/adaptive_preprocessor.py`

```python
import cv2
import numpy as np
from typing import Dict
from .image_analyzer import ImageAnalyzer

class AdaptivePreprocessor:
    def __init__(self):
        self.analyzer = ImageAnalyzer()
    
    def process(self, image_path: str) -> str:
        """自适应预处理"""
        # 分析图像特征
        features = self.analyzer.analyze(image_path)
        
        # 选择策略
        strategy = self._select_strategy(features)
        
        # 应用预处理
        processed_image = self._apply_strategy(image_path, strategy)
        
        # 保存处理后的图像
        output_path = self._save_processed_image(processed_image, image_path)
        
        return output_path
    
    def _select_strategy(self, features: Dict[str, float]) -> Dict[str, str]:
        """根据特征选择预处理策略"""
        strategy = {}
        
        # 去噪策略
        if features['noise_level'] > 0.15:
            strategy['denoise'] = 'strong'
        elif features['noise_level'] > 0.08:
            strategy['denoise'] = 'medium'
        elif features['noise_level'] > 0.03:
            strategy['denoise'] = 'light'
        else:
            strategy['denoise'] = 'none'
        
        # 对比度增强策略
        if features['contrast'] < 0.3:
            strategy['enhance'] = 'strong'
        elif features['contrast'] < 0.6:
            strategy['enhance'] = 'medium'
        else:
            strategy['enhance'] = 'none'
        
        # 锐化策略
        if features['blur_level'] > 0.6:
            strategy['sharpen'] = 'strong'
        elif features['blur_level'] > 0.3:
            strategy['sharpen'] = 'medium'
        else:
            strategy['sharpen'] = 'none'
        
        return strategy
    
    def _apply_strategy(self, image_path: str, strategy: Dict[str, str]) -> np.ndarray:
        """应用预处理策略"""
        image = cv2.imread(image_path)
        if image is None:
            raise ValueError(f"无法读取图像: {image_path}")
        
        processed = image.copy()
        
        # 去噪
        denoise_method = strategy.get('denoise', 'none')
        if denoise_method == 'strong':
            processed = cv2.fastNlMeansDenoisingColored(processed, None, 10, 10, 7, 21)
        elif denoise_method == 'medium':
            processed = cv2.bilateralFilter(processed, 9, 75, 75)
        elif denoise_method == 'light':
            processed = cv2.GaussianBlur(processed, (3, 3), 0)
        
        # 转换为LAB空间进行对比度增强
        lab = cv2.cvtColor(processed, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        
        # CLAHE对比度增强
        enhance_method = strategy.get('enhance', 'none')
        if enhance_method == 'strong':
            clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
        elif enhance_method == 'medium':
            clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        else:
            clahe = None
        
        if clahe:
            l = clahe.apply(l)
        
        processed = cv2.merge([l, a, b])
        processed = cv2.cvtColor(processed, cv2.COLOR_LAB2BGR)
        
        # 锐化
        sharpen_method = strategy.get('sharpen', 'none')
        if sharpen_method != 'none':
            kernel = np.array([[-1, -1, -1],
                              [-1,  9, -1],
                              [-1, -1, -1]])
            if sharpen_method == 'strong':
                kernel = np.array([[-1, -1, -1],
                                  [-1, 16, -1],
                                  [-1, -1, -1]])
            processed = cv2.filter2D(processed, -1, kernel)
        
        return processed
    
    def _save_processed_image(self, image: np.ndarray, original_path: str) -> str:
        """保存处理后的图像"""
        import os
        import uuid
        
        # 生成临时文件名
        temp_dir = os.path.dirname(original_path)
        base_name = os.path.splitext(os.path.basename(original_path))[0]
        output_name = f"{base_name}_processed_{uuid.uuid4().hex[:8]}.jpg"
        output_path = os.path.join(temp_dir, output_name)
        
        # 保存图像
        cv2.imwrite(output_path, image, [cv2.IMWRITE_JPEG_QUALITY, 95])
        
        return output_path
```
